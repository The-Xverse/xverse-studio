async function tabTo(page: Page, target: Locator) {
  await expect(target).toBeVisible();
  for (const [key, limit] of [
    ["Shift+Tab", 40],
    ["Tab", 220],
  ] as const) {
    for (let i = 0; i < limit; i++) {
      if (await target.evaluate((n) => document.activeElement === n)) return;
      await page.keyboard.press(key);
    }
  }
  await expect(target).toBeFocused();
}

const assetPattern = "**/api/v1/assets*";

async function begin(
  page: Page,
  info: TestInfo,
  viewport: { width: number; height: number },
) {
  await page.setViewportSize(viewport);
  await connectStudio(page, info);
  const before = {
    assets: await get(page, "/assets"),
    projects: await get(page, "/projects"),
    runs: await get(page, "/runs"),
  };
  let writes = 0;
  page.on("request", (r) => {
    if (!["GET", "HEAD"].includes(r.method())) writes++;
  });
  return async (details: Record<string, unknown>) => {
    await page.unroute(assetPattern);
    await page.waitForLoadState("networkidle");
    expect({
      assets: await get(page, "/assets"),
      projects: await get(page, "/projects"),
      runs: await get(page, "/runs"),
    }).toEqual(before);
    expect(writes).toBe(0);
    await info.attach("lab-inventory-journey.json", {
      body: JSON.stringify({
        viewport,
        ...details,
        mutativeRequests: writes,
        canonicalListsUnchanged: true,
        noStartOrDeviceOperation: true,
        scope:
          "Native browser UI over actual empty/unadmitted inventory and explicitly labelled controlled public observations; no physical integration claim",
      }),
      contentType: "application/json",
    });
  };
}
async function openLab(page: Page) {
  await activate(
    page,
    page.getByRole("button", { name: "Laboratory", exact: true }),
  );
  await expect(
    lab(page).getByRole("heading", { name: "Physical assets", exact: true }),
  ).toBeVisible();
}
async function controlBounds(
  target: Locator,
  viewport: { width: number; height: number },
) {
  await expect(target).toBeFocused();
  const value = await target.evaluate((n) => {
    const box = n.getBoundingClientRect(),
      style = getComputedStyle(n);
    return {
      box: box.toJSON(),
      outline: style.outlineStyle,
      outlineWidth: parseFloat(style.outlineWidth),
    };
  });
  expect(value.box.left).toBeGreaterThanOrEqual(0);
  expect(value.box.top).toBeGreaterThanOrEqual(0);
  expect(value.box.right).toBeLessThanOrEqual(viewport.width);
  expect(value.box.bottom).toBeLessThanOrEqual(viewport.height);
  expect(value.outline).not.toBe("none");
  expect(value.outlineWidth).toBeGreaterThan(0);
  return value;
}
async function paint(target: Locator, label: string) {
  const result = await renderedContrast(target, label);
  expect.soft(result.supported).toBe(true);
  if (result.supported) expect.soft(result.ratio).toBeGreaterThanOrEqual(4.5);
  return result;
}

      page,
    }, info) => {
      const finish = await begin(page, info, viewport);
      let release!: () => void,
        finished!: () => void,
        ready = false,
        departed = false;
      const gate = new Promise<void>((r) => (release = r)),
        done = new Promise<void>((r) => (finished = r));
      await page.route(assetPattern, async (route) => {
        try {
          const actual = status === 200 ? await route.fetch() : undefined;
          if (actual) expect(actual.status()).toBe(200);
          ready = true;
          await gate;
          if (actual) await route.fulfill({ response: actual });
          else
            await route.fulfill({
              status: 503,
              contentType: "application/json",
              body: JSON.stringify({
                code: "INVENTORY_UNAVAILABLE",
                message: "Synthetic departed inventory failure",
              }),
            });
        } catch (error) {
          if (!departed) throw error;
        } finally {
          finished();
        }
      });
      try {
        await openLab(page);
        await expect.poll(() => ready).toBe(true);
        await expect(refresh(page)).toBeDisabled();
        await expect(lab(page).getByRole("status")).toHaveText(
          "Reading inventory observations…",
        );
        const home = page.getByRole("button", {
          name: "X-Verse Studio home",
          exact: true,
        });
        await activate(page, home);
        departed = true;
        await expect(lab(page)).toHaveCount(0);
        release();
        await done;
        await page.waitForLoadState("networkidle");
        await expect(home).toBeFocused();
        await expect(lab(page)).toHaveCount(0);
        await expect(page.locator(".alerts [role=alert]")).toHaveCount(0);
        await finish({
          status,
          departed: true,
          actualResponseHeld: status === 200,
          labelledTransportFailure: status === 503,
        });
      } finally {
        departed = true;
        release();
        await done;
        await page.unroute(assetPattern);
      }
    });
  }

      }
    });

        "observation is complete",
      );

      await page.keyboard.press("Enter");

      expect(reads).toBe(2);
      const home = page.getByRole("button", {
        name: "X-Verse Studio home",
        exact: true,
      });
      await tabTo(page, home);
      release();
      await done;
      await expect(lab(page).getByRole("status")).toContainText(
        "observation is complete",
      );
      await expect(home).toBeFocused();
      expect(reads).toBe(2);
      await finish({
        reads,
        pendingRepeatCreatedNoExtraRead: true,
        deliberatelyMovedFocusPreserved: true,
      });
    } finally {
      release();
      if (reads > 1) await done;
      await page.unroute(assetPattern);
    }
  });

async function activate(page: Page, target: Locator) {
  await tabTo(page, target);
  await page.keyboard.press("Enter");
}
async function get(page: Page, route: string) {
  return page.evaluate(async (route) => {
    const response = await fetch("/api/v1" + route);
    if (!response.ok) throw new Error(`Owned read failed: ${response.status}`);
    return response.json();
  }, route);
}
