def read(path):
    return json.loads(path.read_text())

def clean_text(value):
    for original, replacement in [(str(root.parent/'.xverse-studio-cache'), '<loop-cache>'),
            (str(root), '<repo>'), ('<repo>', '<repo>'),
            (sys.base_prefix, '<python-install>'), ('<repo>', '<repo>')]:
        value = value.replace(original, replacement)
    return value

def clean(value):
    if isinstance(value, str):
        return clean_text(value)
    if isinstance(value, list):
        return [clean(v) for v in value]
    if isinstance(value, dict):
        result = {k: clean(v) for k, v in value.items()}
        if value.get('contentType', '').startswith(('application/json', 'text/')) and 'body' in value:
            try:
                result['body'] = base64.b64encode(clean_text(base64.b64decode(value['body']).decode()).encode()).decode()
            except (ValueError, UnicodeError):
                pass
        return result
    return value

def save(name, value):
    p = out/name
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(clean(value), indent=2)+'\n')

def copy_text(name, source):
    p = out/name
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text('\n'.join(v.rstrip() for v in clean_text(source.read_text()).splitlines()).rstrip()+'\n')

def results(report):
    for suite in report.get('suites', []):
        yield from results(suite)
    for spec in report.get('specs', []):
        for test in spec['tests']:
            for result in test['results']:
                yield spec['title'], result

for stage in ['initial-red', 'first-four-successor', 'thirty-before-review', 'before-physical-review-targeted', 'review-interrupted-full', 'physical-declaration-red', 'before-unicode-targeted', 'before-unicode-full', 'before-unicode-performance', 'unicode-red']:
    path = local/stage
    for name in ['summary.json', 'source-before.json', 'source-after.json', 'browser-report.json']:
        save(f'{stage}/{name}', read(path/name))
    for name in ['classification.json']:
        if (path/name).exists():
            save(f'{stage}/{name}', read(path/name))
    for name, dest in [('verify.py', 'controller.py.txt'), ('test.ts.txt', 'test.ts.txt'),
            ('browser.log', 'browser.txt'), ('product.patch', 'product.patch')]:
        if (path/name).exists():
            copy_text(f'{stage}/{dest}', path/name)
    for p in (path/'browser-output').glob('**/error-context.md'):
        copy_text(f'{stage}/contexts/{p.parent.name}/{p.name}', p)
    for title, result in results(read(path/'browser-report.json')):
        for a in result.get('attachments', []):
            if a.get('contentType') == 'application/json' and 'body' in a:
                view = re.search(r'(\d+x\d+)', title)
                viewport = view.group(1) if view else 'multiple-or-unspecified'
                mode = title.split(': ', 1)[-1].rsplit(' at ', 1)[0].replace(' ', '-')
                save(f'{stage}/journeys/{viewport}/{mode}/{a["name"]}', json.loads(base64.b64decode(a['body'])))
