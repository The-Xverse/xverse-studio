from pathlib import Path
import json
from copy import deepcopy
from xverse_studio.lab import LabRegistry
from xverse_studio.lab.assets import inventory_observation

class SyntheticInventory:
    def __init__(self, failure=False, unclassified=False, unicode_boundary=False):
        self.failure = failure
        self.unclassified = unclassified
        self.unicode_boundary = unicode_boundary

    def inventory(self):
        if self.failure:
            raise OSError('Synthetic inventory failure')
        asset = deepcopy({
            'assetId': 'urn:fixture:inventory-presentation', 'providerId': 'fixture',
            'providerVersion': 'fixture-provider-1.0', 'physical': True,
            'identity': {'serial': 'SYNTHETIC-UI-001', 'model': 'Inventory contract fixture'},
            'firmware': {'status': 'observed', 'version': 'fixture-1.2.3'},
            'artifactRevision': {'status': 'unknown'},
            'interfaces': {'command': {'id': 'command', 'direction': 'input', 'payloadSchema': 'urn:fixture:command'}},
            'supportedOS': ['Linux'], 'available': True,
            'capabilities': ['observe', 'stimulate'], 'evidenceScope': 'fixture',
        })
        if self.unclassified:
            asset.pop('physical')
        if self.unicode_boundary:
            asset['identity']['model'] = '🧪' * 256
            asset['firmware']['version'] = '🧪' * 256
        return [asset]

    def __getattr__(self, name):
        raise AssertionError(f'This fixture must never delegate {name}')

def response(failure, unclassified=False, unicode_boundary=False):
    value = inventory_observation(LabRegistry({'fixture': SyntheticInventory(failure, unclassified, unicode_boundary)}))
    return {**value, 'nextCursor': None, 'scope': 'explicitly_registered_inventory_observations',
        'physicalProviderStatus': 'unavailable', 'activeReady': False,
        'diagnostics': [{'code': 'PHYSICAL_PROVIDER_NOT_ADMITTED'}, *value['diagnostics']]}

out = Path('tests/fixtures/providers/lab-inventory-observation.json')
value = {'fixtureScope': 'UI transport fixture generated by the real public projection over an owned in-memory provider. No device or live provider admission.',
    'observed': response(False), 'incomplete': response(True), 'unclassified': response(False, True),
    'unclassifiedSource': {'physicalPropertyPresent': False, 'scope': 'Actual backend projection of an omitted physical property; false does not establish a virtual realization'},
    'unicode': response(False, unicode_boundary=True),
    'unicodeSource': {'unicodeCodePoints': 256, 'utf16CodeUnits': 512, 'scope': 'Real public projection accepts bounded Unicode code points, not a narrower UTF16-code-unit limit'}}
assert value['observed']['complete'] and not value['incomplete']['complete']
assert not value['observed']['items'][0]['reservable']
assert value['unclassified']['items'][0]['physical'] is False
assert len(value['unicode']['items'][0]['firmware']['version']) == 256
assert len(value['unicode']['items'][0]['firmware']['version'].encode('utf-16-le')) // 2 == 512
out.write_text(json.dumps(value, indent=2)+'\n')
print('Generated public-shape inventory fixture; no active provider or device operation')
