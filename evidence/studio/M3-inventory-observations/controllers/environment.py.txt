from pathlib import Path
import hashlib
import json
import os
import subprocess
import sys
import psutil

root = Path.cwd()
cache = root.parent/'.xverse-studio-cache'
out = root/'.studio-local/lab-inventory'
env = dict(os.environ, PATH=str(cache/'node-v24.22.1-linux-x64/bin')+os.pathsep+os.environ['PATH'],
    TMPDIR=str(cache/'tmp'), TEMP=str(cache/'tmp'), TMP=str(cache/'tmp'),
    XDG_CACHE_HOME=str(cache/'chromium-metadata'), PLAYWRIGHT_BROWSERS_PATH=str(cache/'playwright'),
    PYTEST_DISABLE_PLUGIN_AUTOLOAD='1')
subprocess.run([sys.executable, '-I', 'scripts/record_environment.py', '--output', str(out/'environment.json')],
    env=env, stdout=subprocess.DEVNULL, check=True)
pipelines = []
for process in psutil.process_iter(['pid', 'cmdline']):
    try:
        if any(arg == '.studio-local/lab-inventory/pipeline.py' for arg in process.info['cmdline'] or []):
            if Path(process.cwd()).resolve() == root.resolve():
                pipelines.append(process)
    except (psutil.NoSuchProcess, psutil.AccessDenied):
        continue
assert len(pipelines) == 1
executables = set()
for child in pipelines[0].children(recursive=True):
    try:
        path = Path(child.exe())
        if path.is_relative_to(cache/'playwright') and path.name in ['chrome', 'headless_shell', 'chrome-headless-shell']:
            executables.add(path)
    except (psutil.NoSuchProcess, psutil.AccessDenied):
        continue
assert executables, 'No live owned browser descendant found; retry the same running pipeline observation'
versions = []
for path in sorted(executables):
    value = subprocess.check_output([str(path), '--version'], env=env, text=True).strip()
    versions.append({'executable': str(path), 'version': value, 'sha256': hashlib.sha256(path.read_bytes()).hexdigest()})
p = out/'environment.json'
value = json.loads(p.read_text())
value['observedBrowserExecutables'] = versions
value['browserObservationScope'] = 'Read-only executable/version/hash observation of descendants of this exact live owned inventory pipeline; no private arguments or authentication state collected'
p.write_text(json.dumps(value, indent=2)+'\n')
print(json.dumps({'observedBrowserVersions': [v['version'] for v in versions], 'executables': len(versions)}))
