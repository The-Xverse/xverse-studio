import { test, expect } from "../../../node_modules/@playwright/test";
import { connectStudio } from "../../../tests/e2e/connect-studio";
import { readFileSync, writeFileSync } from "node:fs";
import path from "node:path";

test("newer modal supersedes recovery with no rendered project", async ({ page }, info) => {
  await connectStudio(page, info);
  const project = await page.evaluate(async () => {
    const { csrfToken } = await fetch("/api/v1/session").then(r => r.json());
    const response = await fetch("/api/v1/projects", { method: "POST", headers: { "Content-Type": "application/json", "X-CSRF-Token": csrfToken }, body: JSON.stringify({ name: "No project recovery modal probe" }) });
    if (response.status !== 201) throw new Error("Owned project creation refused");
    const value = await response.json(); localStorage.setItem("studio.projectId", value.projectId); return value;
  });
  const document = Object.values(project.documents).find((d: any) => d.path.endsWith("/system.xdl.yaml")) as any;
  const file = path.resolve(process.env.STUDIO_BROWSER_DATA!, "projects", project.projectId, "revisions", String(project.revision), document.path);
  expect(file.startsWith(path.resolve(process.env.STUDIO_BROWSER_DATA!) + path.sep)).toBe(true);
  const original = readFileSync(file); writeFileSync(file, Buffer.from([0xff, 0xfe, 0x00, 0x62]));
  let release!: () => void, committed!: () => void;
  const deliver = new Promise<void>(resolve => { release = resolve; });
  const ready = new Promise<void>(resolve => { committed = resolve; });
  await page.route(`**/api/v1/projects/${project.projectId}/recoveries`, async route => {
    if (route.request().method() !== "POST") return route.continue();
    const response = await route.fetch(), body = await response.text(); expect(response.status()).toBe(200);
    committed(); await deliver; await route.fulfill({ response, body });
  });
  try {
    await page.reload();
    const recovery = page.getByRole("group", { name: "Source recovery", exact: true });
    await expect(recovery).toBeVisible();
    await expect(page.getByRole("heading", { name: project.name, exact: true })).toHaveCount(0);
    await recovery.getByRole("button", { name: `Restore committed revision ${project.revision}`, exact: true }).focus();
    await page.keyboard.press("Enter"); await ready;
    const create = page.getByRole("button", { name: "New project", exact: true });
    await create.focus(); await page.keyboard.press("Enter");
    await expect(page.getByRole("dialog", { name: "Create a System workspace", exact: true })).toBeVisible();
    await page.keyboard.press("Escape");
    await expect(page.getByRole("dialog")).toHaveCount(0);
    const reply = page.waitForResponse(r => r.url().endsWith(`/projects/${project.projectId}/recoveries`) && r.request().method() === "POST");
    release(); expect((await reply).status()).toBe(200); await page.waitForLoadState("networkidle");
    await expect(page.getByRole("heading", { name: "Start with an example", exact: true })).toBeVisible();
    await expect(page.getByRole("heading", { name: project.name, exact: true })).toHaveCount(0);
    expect((await (await page.request.get(`/api/v1/projects/${project.projectId}`)).json()).revision).toBe(project.revision + 1);
  } finally { release(); await page.unrouteAll({ behavior: "wait" }); writeFileSync(file, original); }
});
