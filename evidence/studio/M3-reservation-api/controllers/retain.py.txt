read = lambda p: json.loads(p.read_text())

def clean_text(value):
    for original, replacement in [(str(root.parent / '.xverse-studio-cache'), '<loop-cache>'),
        (str(root), '<repo>'), (str(Path.home() / 'xverse-studio'), '<repo>'),
        (sys.base_prefix, '<python-install>'), (str(Path.home()), '<user-home>'),
        (str(root.parent), '<workspace-volume>')]:
        value = value.replace(original, replacement)
    value = re.sub(r'(?i)(\bstudio_session=)[A-Za-z0-9_-]{16,}', r'\1<redacted>', value)
    value = re.sub(r'''(?i)(x-csrf-token["']?\s*[:=]\s*["']?)[A-Za-z0-9._-]{16,}''', r'\1<redacted>', value)
    return re.sub(r'(?i)(bearer\s+)[A-Za-z0-9._-]{16,}', r'\1<redacted>', value)

def clean(value):
    if isinstance(value, str):
        return clean_text(value)
    if isinstance(value, list):
        return [clean(v) for v in value]
    if isinstance(value, dict):
        result = {key: '<redacted>' if key.lower() in {'studio_session', 'csrftoken', 'x-csrf-token',
            'bootstraptoken', 'x-studio-bootstrap', 'sessiontoken', 'ownerkey', 'authorization', 'cookie', 'set-cookie'} else clean(v) for key, v in value.items()}
        if value.get('contentType', '').startswith(('text/', 'application/json')) and 'body' in value:
            try:
                result['body'] = base64.b64encode(clean_text(base64.b64decode(value['body']).decode()).encode()).decode()
            except (ValueError, UnicodeError):
                pass
        return result
    return value

def save(name, value):
    path = out / name
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(clean(value), indent=2) + '\n')

def copy(name, path):
    if path.suffix == '.json':
        return save(name, read(path))
    target = out / name
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text('\n'.join(line.rstrip() for line in clean_text(path.read_text()).splitlines()).rstrip() + '\n')

def totals(tree):
    return {key: sum(int(e.get(key, 0)) for e in tree.getroot().iter('testsuite')) for key in ('tests', 'failures', 'errors', 'skipped')}

xml_records = []
def xml(name, source, expected=None):
    tree = ET.parse(source)
    values = totals(tree)
    if expected is not None:
        assert values == expected, (name, values)
    identities = [(e.get('classname'), e.get('name')) for e in tree.getroot().iter('testcase')]
    removed = 0
    for element in tree.getroot().iter():
        if 'hostname' in element.attrib:
            del element.attrib['hostname']
            removed += 1
        element.attrib.update({key: clean_text(value) for key, value in element.attrib.items()})
        if element.text:
            element.text = '\n'.join(line.rstrip() for line in clean_text(element.text).split('\n'))
        if element.tail:
            element.tail = clean_text(element.tail)
    target = out / name
    target.parent.mkdir(parents=True, exist_ok=True)
    tree.write(target, encoding='utf-8', xml_declaration=True)
    assert totals(ET.parse(target)) == values
    assert identities == [(e.get('classname'), e.get('name')) for e in tree.getroot().iter('testcase')]
    xml_records.append({'path': name, **values, 'hostnameAttributesRemoved': removed,
        'pathsAndCredentialLiteralsSanitizedBeforeSerialization': True, 'failureTextTrailingWhitespaceRemoved': True,
        'testcaseIdentitiesAndOutcomesPreserved': True,
        'privateRawSha256': hashlib.sha256(source.read_bytes()).hexdigest(),
        'retainedSha256': hashlib.sha256(target.read_bytes()).hexdigest()})
