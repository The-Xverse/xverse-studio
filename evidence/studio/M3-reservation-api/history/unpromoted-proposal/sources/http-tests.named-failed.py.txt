"""Proposed facade against actual local session/Origin/CSRF middleware.

Handlers are explicitly replaced in this owned test app. This is not evidence
that production has positive reservations, nor actual provider admission.
"""
import importlib
import importlib.util
from pathlib import Path
import sys
from types import ModuleType, SimpleNamespace
import pytest
from fastapi.testclient import TestClient

from xverse_studio.api.app import create_app

package = ModuleType('owned_lab_http_proposal')
package.__path__ = [str(Path(__file__).parent)]
sys.modules[package.__name__] = package
operations = importlib.import_module(package.__name__ + '.operations')
http = importlib.import_module(package.__name__ + '.lab_http')
spec = importlib.util.spec_from_file_location('owned_lab_http_fixture', Path(__file__).resolve().parents[3] / 'tests/fixtures/providers/physical/provider.py')
fixtures = importlib.util.module_from_spec(spec)
spec.loader.exec_module(fixtures)


def prepare(root, provider, policy, now):
    app = create_app(root, allowed_hosts=['testserver'], bootstrap_token='owned-http-fixture-bootstrap')
    service = operations.LabOperations(app.state.studio.store, {provider.provider_id: provider},
        {provider.provider_id: 'sha256:' + 'a' * 64}, policy, clock=lambda: now[0])
    replaced = {'/api/v1/assets/{assetId}/reservations', '/api/v1/reservations/{reservationId}',
                '/api/v1/reservations/{reservationId}/renewals'}
    prior = [route for route in app.router.routes if getattr(route, 'path', None) in replaced]
    assert len(prior) == 4
    app.router.routes[:] = [route for route in app.router.routes if route not in prior]
    http.install_proposal_routes(app, service)
    return app, service


def login(client):
    client.headers['Origin'] = 'http://testserver'
    result = client.post('/api/v1/session', json={'bootstrapToken': 'owned-http-fixture-bootstrap'})
    assert result.status_code == 200
    client.headers['X-CSRF-Token'] = result.json()['csrfToken']


@pytest.fixture
def bench(tmp_path):
    now = [100]
    provider = fixtures.PhysicalContractFixture(lambda: now[0])
    policy = operations.LabAuthority({('local-operator', provider.asset_id): {'observe'}})
    app, service = prepare(tmp_path / 'owned proposed HTTP Ω', provider, policy, now)
    with TestClient(app, base_url='http://testserver') as client:
        login(client)
        yield SimpleNamespace(client=client, service=service, provider=provider, policy=policy, now=now)


def acquire(bench, *, key='acquire', body=None):
    return bench.client.post('/api/v1/assets/' + bench.provider.asset_id + '/reservations',
        headers={'Idempotency-Key': key}, json={} if body is None else body)


def test_real_session_positive_observe_same_key_conflict_and_release(bench):
    first = acquire(bench)
    assert first.status_code == 202 and first.json()['status'] == 'completed'
    assert acquire(bench).json() == first.json()
    rid = first.json()['reservationId']
    record = bench.client.get('/api/v1/reservations/' + rid)
    assert record.status_code == 200 and record.json()['scope'] == 'observe' and record.json()['evidenceScope'] == 'fixture'
    op = bench.client.get('/api/v1/operations/' + first.json()['operationId'])
    assert op.status_code == 200 and op.json() == first.json()
    conflict = acquire(bench, body={'durationSeconds': 31})
    assert conflict.status_code == 409 and conflict.json()['code'] == 'IDEMPOTENCY_CONFLICT'
    released = bench.client.delete('/api/v1/reservations/' + rid, headers={'Idempotency-Key': 'release'})
    assert released.status_code == 202 and released.json()['result']['status'] == 'released'
    assert bench.provider.calls.count('reserve') == bench.provider.calls.count('release') == 1


@pytest.mark.parametrize('scope', ['stimulate', 'destructive'])
def test_each_active_scope_requires_its_own_server_grant(bench, scope):
    refused = acquire(bench, body={'scope': scope})
    assert refused.status_code == 403 and refused.json()['code'] == 'FORBIDDEN'
    assert bench.provider.calls.count('reserve') == 0
    bench.policy.set_scopes('local-operator', bench.provider.asset_id, {scope})
    accepted = acquire(bench, body={'scope': scope})
    assert accepted.status_code == 202 and accepted.json()['status'] == 'completed'
    record = bench.client.get('/api/v1/reservations/' + accepted.json()['reservationId']).json()
    assert record['scope'] == scope and record['evidenceScope'] == 'fixture'
    other = 'destructive' if scope == 'stimulate' else 'stimulate'
    still_refused = acquire(bench, key='other', body={'scope': other})
    assert still_refused.status_code == 403 and bench.provider.calls.count('reserve') == 1


@pytest.mark.parametrize('body', [{'owner': 'different-owner'}, {'grantedScopes': ['stimulate']},
                                 {'providerId': 'unapproved'}, {'scope': 'admin'},
                                 {'durationSeconds': True}, {'durationSeconds': 301}])
def test_browser_cannot_supply_owner_grants_provider_or_invalid_limits(bench, body):
    result = acquire(bench, body=body)
    assert result.status_code == 422 and result.json()['code'] == 'REQUEST_INVALID'
    assert bench.provider.calls == [] and bench.service.store.all('operation') == []


@pytest.mark.parametrize('header,value', [('X-CSRF-Token', 'bad-fixture-csrf'), ('Origin', 'http://foreign.invalid')])
def test_actual_boundary_refuses_before_provider_contact(bench, header, value):
    bench.client.headers[header] = value
    result = acquire(bench)
    assert result.status_code == 403 and result.json()['code'] == 'FORBIDDEN'
    assert bench.provider.calls == []


def test_unknown_response_is_accepted_without_reacquisition_on_retry(bench):
    bench.provider.lose_reserve_ack = True
    first = acquire(bench)
    assert first.status_code == 202 and first.json()['status'] == 'unknown'
    assert acquire(bench).json() == first.json() and bench.provider.calls.count('reserve') == 1
    rid = first.json()['reservationId']
    record = bench.client.get('/api/v1/reservations/' + rid)
    assert record.json()['status'] == 'unknown' and record.json()['fencingIdentity'] is None
    renewed = bench.client.post('/api/v1/reservations/' + rid + '/renewals',
        headers={'Idempotency-Key': 'renew-unknown'}, json={'fencingIdentity': 'invented-fixture-fence'})
    assert renewed.status_code == 422 and renewed.json()['code'] == 'OWNERSHIP_UNCERTAIN'
    assert bench.provider.calls.count('renew') == 0
