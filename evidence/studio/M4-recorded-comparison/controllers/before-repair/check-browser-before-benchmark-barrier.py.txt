from pathlib import Path
import os,sys,json,subprocess,time,hashlib,shutil
root=Path.cwd().resolve(); cache=root.parent/'.xverse-studio-cache'
name,pattern=sys.argv[1:3]
out=root/'.studio-local/comparison-workflow'/name; out.mkdir(exist_ok=False)
env=dict(os.environ, PATH=str(cache/'node-v24.22.1-linux-x64/bin')+os.pathsep+os.environ['PATH'],TMPDIR=str(cache/'tmp'),TEMP=str(cache/'tmp'),TMP=str(cache/'tmp'),UV_CACHE_DIR=str(cache/'uv'),npm_config_cache=str(cache/'npm'),PLAYWRIGHT_BROWSERS_PATH=str(cache/'playwright'),PYTEST_DISABLE_PLUGIN_AUTOLOAD='1')
def inputs():
    paths=subprocess.check_output(['git','ls-files','--cached','--others','--exclude-standard'],text=True).splitlines()
    paths += [str(p.relative_to(root)) for p in (root/'apps/studio-web/dist').rglob('*') if p.is_file()]
    return {p:hashlib.sha256((root/p).read_bytes()).hexdigest() for p in sorted(set(paths)) if (root/p).is_file() and (p.startswith(('apps/','services/','scripts/','tests/','packages/','config/')) or p in ['pyproject.toml','uv.lock','package.json','package-lock.json','docs/architecture/ADR-0019-scenario-evidence-timeline.md','docs/architecture/ADR-0020-campaign-workflow.md','docs/architecture/ADR-0021-recorded-comparison.md','docs/architecture/ADR-0022-owned-worker-state-handoff.md'])}
before=inputs(); (out/'source-before.json').write_text(json.dumps(before,indent=2))
command=[str(root/'.venv/bin/python'),'-I','scripts/test_browser.py'] + (['--performance','--grep','operability:'] if pattern == 'PERFORMANCE' else ['--grep',pattern] if pattern != 'ALL' else [])
start=time.monotonic()
with (out/'browser.log').open('w') as log: result=subprocess.run(command,env=env,stdout=log,stderr=subprocess.STDOUT)
after=inputs(); (out/'source-after.json').write_text(json.dumps(after,indent=2))
for name in ['browser-report.json','browser']:
    src=root/'test-results'/name
    if src.is_dir(): shutil.copytree(src,out/'browser-output')
    elif src.is_file(): shutil.copy2(src,out/name)
summary={'baseCandidate':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),'command':command,'exitCode':result.returncode,'seconds':round(time.monotonic()-start,3),'sourceUnchanged':before==after}
(out/'summary.json').write_text(json.dumps(summary,indent=2));print(json.dumps(summary),flush=True)
sys.exit(result.returncode or int(before!=after))
