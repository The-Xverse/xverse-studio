import { test, expect } from '@playwright/test';
import { connectStudio } from '../../tests/e2e/connect-studio';
import { traceApi } from '../../tests/e2e/trace-claims-flow';

for (const width of [1024, 1440]) {
  test(`actual rerun acknowledgement preserves a newer Model view at ${width}`, async ({page}, info) => {
    await page.setViewportSize({width, height:width === 1024 ? 768 : 900});
    await connectStudio(page, info);
    let project = (await traceApi(page, '/projects', 'POST', {name:'Owned rerun view probe '+width})).body;
    const edited = await traceApi(page, `/projects/${project.projectId}/edits`, 'POST', {
      baseRevision:project.revision, edits:[{kind:'set_parameter',parameterId:'duration',value:0.2}],
    },project.revision);
    expect(edited.status).toBe(200); project=edited.body;
    const pf = (await traceApi(page, `/projects/${project.projectId}/preflights`, 'POST', {revision:project.revision})).body;
    expect(pf.status).toBe('ready');
    const accepted = await traceApi(page,'/runs','POST',{snapshotId:pf.snapshotId,preflightId:pf.preflightId});
    expect(accepted.status).toBe(202); const original=accepted.body.runId;
    await expect.poll(async ()=>(await traceApi(page,'/runs/'+original)).body.state,{timeout:20000}).toBe('stopped');
    await page.evaluate(({pid,rid})=>{
      localStorage.setItem('studio.projectId',pid);
      localStorage.setItem('studio.lastRun',rid);
      localStorage.setItem('studio.lastRun.'+pid,rid);
    },{pid:project.projectId,rid:original});
    await page.reload();
    await page.getByRole('tab',{name:'Results',exact:false}).click();
    await expect(page.getByLabel('Recorded run',{exact:true})).toHaveValue(original);
    let release!:()=>void, ready!:()=>void, completed!:()=>void;
    const held=new Promise<void>(resolve=>release=resolve);
    const actual=new Promise<void>(resolve=>ready=resolve);
    const delivered=new Promise<void>(resolve=>completed=resolve);
    let acknowledged:any;
    await page.route(`**/api/v1/runs/${original}/reruns`,async route=>{
      const response=await route.fetch();
      acknowledged=await response.json();
      expect(response.status()).toBe(202);
      ready(); await held;
      await route.fulfill({response}); completed();
    });
    const rerun=page.getByRole('button',{name:'Rerun frozen inputs',exact:true});
    await rerun.focus(); await page.keyboard.press('Enter');
    await actual;
    const operation=(await traceApi(page,'/operations/'+acknowledged.operationId)).body;
    expect(operation.status).toBe('completed');
    const child=operation.result.runId; expect(child).not.toBe(original);
    await expect.poll(async ()=>(await traceApi(page,'/runs/'+child)).body.state,{timeout:20000}).toBe('stopped');
    await page.getByRole('tab',{name:'Model',exact:true}).click();
    const focus=page.getByRole('button',{name:'Parameters',exact:true});
    await focus.focus();
    expect(await page.evaluate(()=>localStorage.getItem('studio.lastRun'))).toBe(original);
    release(); await delivered;
    await expect.poll(async()=>page.evaluate(()=>localStorage.getItem('studio.pendingRerun.'+localStorage.getItem('studio.lastRun'))),{timeout:10000}).toBeNull();
    await info.attach('actual-rerun-view-context',{contentType:'application/json',body:JSON.stringify({
      scope:'Actual native owned rerun; only its real HTTP202 delivery was held. No provider or evidence fixture substituted.',
      originalRunId:original,childRunId:child,operationId:acknowledged.operationId,
      intendedView:'Model',currentView:await page.getByRole('tab',{name:'Model',exact:true}).getAttribute('aria-selected'),
      currentRunId:await page.evaluate(()=>localStorage.getItem('studio.lastRun')),
    })});
    await expect(page.getByRole('tab',{name:'Model',exact:true})).toHaveAttribute('aria-selected','true');
    await expect(focus).toBeFocused();
    expect(await page.evaluate(()=>localStorage.getItem('studio.lastRun'))).toBe(original);
  });
}
