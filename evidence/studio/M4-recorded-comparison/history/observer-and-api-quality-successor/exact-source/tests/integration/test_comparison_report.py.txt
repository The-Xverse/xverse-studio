def fixture_run():

    resources = [
        {"kind": kind, "uri": "xdl://fixture/" + kind.lower() + "/example", "version": "1.0", "semanticDigest": {"algorithm": "sha256", "value": "1" * 64}}
        for kind in ["System", "Deployment", "Scenario", "Profile", "Component"]
    ]
    snapshot = {
        "snapshotId": "fixture-snapshot", "snapshotDigest": "sha256:" + "2" * 64,
        "projectId": "fixture-project", "revision": 1,
        "semanticDigest": "sha256:" + "3" * 64, "planDigest": "sha256:" + "4" * 64,
        "configuration": {"gain": 1},
        "selection": {"systemRef": resources[0]["uri"], "deploymentRef": resources[1]["uri"], "scenarioRef": resources[2]["uri"]},
        "providerId": "fixture-provider", "providerIdentity": {"providerId": "fixture-provider", "providerVersion": "1.0", "contractVersion": "1.0", "implementationDigest": "sha256:" + "5" * 64},
        "capabilityDigest": "sha256:" + "6" * 64, "compiler": {"id": "fixture-compiler", "version": "1.0"}, "profileLocks": [],
        "plan": {"provenance": {"resources": resources}, "profile": {"id": "fixture"}, "seed": None,
                 "observers": [{"observerId": "fixture-observer", "timeDomainId": "simulation-time", "payloadPolicy": {"unitSemantics": "1"}}],
                 "bindings": [{"id": "fixture-binding", "artifactRefs": [{"id": "fixture-model", "version": "1.0", "digest": "sha256:" + "7" * 64}]}],
                 "timeDomains": [{"id": "simulation-time", "clockClass": "simulation", "epoch": "0", "rate": "1", "monotonic": True, "canonicalUnit": "tick"}], "timeMappings": []},
    }
    return {"runId": "fixture-run", "snapshot": snapshot, "snapshotIntegrity": "valid", "state": "stopped", "cleanup": "complete", "effectiveInputsMatch": True,
            "evidence": {"integrity": "valid", "status": "complete"},
            "observations": [{"observerRef": "fixture-observer", "field": "state", "unit": "1", "quantity": "dimensionless", "unitDefinitionDigest": "sha256:" + "8" * 64,
                              "clockDomainRef": "simulation-time", "sourceTime": {"resolution": "microsecond", "uncertainty": 0}, "metadataConsistent": True, "identityVerified": True,
                              "coverage": {"verified": True, "complete": True, "startTick": "0", "endTick": "10", "gaps": []},
                              "summary": {"status": "available", "sampleCount": 2, "firstTick": "0", "lastTick": "5", "minimum": 0.0, "maximum": 1.0, "lastValue": 1.0}}]}


def test_report_preserves_frozen_provenance_and_distinct_result_dimensions():
    left = fixture_run(); right = deepcopy(left); right['runId'] = 'other'
    right['snapshot']['configuration']['gain'] = 2
    before = deepcopy((left, right)); report = compare_runs(left, right)
    assert report['comparisonVersion'] == '1.0'
    assert report['compatible'] is True
    assert report['left']['snapshotId'] == 'fixture-snapshot'
    assert report['left']['cleanup'] == 'complete'
    assert report['left']['evidenceIntegrity'] == 'valid'
    rows = {r['field']: r for r in report['configuration']}
    assert rows['parameters']['status'] == 'different'
    assert rows['providerIdentity']['left']['providerVersion'] == '1.0'
    assert rows['artifactIdentities']['left'][0]['artifactRefs'][0]['digest'] == 'sha256:' + '7' * 64
    assert rows['clockDefinitions']['left']['domains'][0]['epoch'] == '0'
    assert report['fields'][0]['left']['summary']['lastValue'] == 1.0
    assert report['rerunRequiresFreshAdmission'] is True
    assert report['mode'] == 'recorded_observation_comparison'
    assert 'verdict' not in report and 'verdict' not in report['left']
    assert (left, right) == before

def test_signed_ticks_and_large_integer_precision_are_preserved():
    left = fixture_run(); right = deepcopy(left)
    for side in (left, right):
        side['observations'][0]['coverage'].update(startTick='-20', endTick='-10')
        side['observations'][0]['summary'].update(firstTick='-20', lastTick='-15')
    assert compare_runs(left, right)['compatible'] is True
    for side in (left, right): side['observations'][0]['summary'].update(firstTick='0', lastTick='5')
    left['observations'][0]['coverage'].update(startTick='0', endTick='9007199254740993')
    right['observations'][0]['coverage'].update(startTick='0', endTick='9007199254740994')
    assert compare_runs(left, right)['compatible'] is False


@pytest.mark.parametrize('change', ['observer', 'clock-definition', 'uncertainty', 'metadata', 'coverage', 'effective-inputs'])
def test_unproven_observer_time_or_source_quality_blocks_pairing(change):
    left = fixture_run(); right = deepcopy(left)
    if change == 'observer':
        del right['observations'][0]['observerRef']
    elif change == 'clock-definition':
        right['snapshot']['plan']['timeDomains'][0]['rate'] = '2'
    elif change == 'uncertainty':
        right['observations'][0]['sourceTime']['uncertainty'] = .1
    elif change == 'metadata':
        right['observations'][0]['metadataConsistent'] = False
    elif change == 'coverage':
        right['observations'][0]['coverage']['gaps'] = [{'reason': 'labelled missing sample'}]
    else:
        right['effectiveInputsMatch'] = False
    assert compare_runs(left, right)['compatible'] is False

def test_public_report_uses_frozen_provider_artifacts_clocks_and_real_summaries(client):
    from test_advanced_api import prepared, start, wait_run
    _, pf = prepared(client); rid = start(client, pf, key='recorded-report')
    assert wait_run(client, rid)['state'] == 'stopped'
    store = client.app.state.studio.store
    snapshot = store.get('snapshot', pf['snapshotId'])
    protected = {p: hashlib.sha256(p.read_bytes()).hexdigest() for p in [store.root/'runs'/rid/'evidence.jsonl', store.root/'snapshots'/pf['snapshotId']/'snapshot.json']}
    report = client.post('/api/v1/comparisons', json={'leftRunId': rid, 'rightRunId': rid})
    assert report.status_code == 200, report.text
    out = report.json(); assert out['comparisonVersion'] == '1.0'
    assert out['compatible'] is True, out['diagnostics']
    rows = {r['field']: r for r in out['configuration']}
    assert rows['providerIdentity']['left'] == snapshot['providerIdentity']
    assert rows['clockDefinitions']['left']['domains'] == snapshot['plan']['timeDomains']
    assert rows['artifactIdentities']['left'] == [{'bindingId': b['id'], 'artifactRefs': b['artifactRefs']} for b in snapshot['plan']['bindings']]
    assert rows['profileIdentities']['left']['resources'] == [r for r in snapshot['plan']['provenance']['resources'] if r['kind'] == 'Profile']
    assert all(f['left']['summary']['sampleCount'] > 0 and f['left']['summary']['status'] == 'available' for f in out['fields'])
    assert client.get(f'/api/v1/runs/{rid}/comparison', params={'baselineRunId': rid}).json() == out
    assert all(hashlib.sha256(p.read_bytes()).hexdigest() == h for p,h in protected.items())

def test_reference_projection_rejects_mixed_metadata_and_preserves_explicit_gaps():
    from xverse_studio.comparison import recorded_input
    snapshot = fixture_run()['snapshot']
    run = {'runId': 'fixture-run', 'providerId': snapshot['providerId'], 'state': 'stopped', 'cleanup': 'complete', 'evidence': 'complete',
           'effectivePlanDigest': snapshot['planDigest'], 'effectiveSemanticDigest': snapshot['semanticDigest']}
    def observation(tick, unit='1', clock='simulation-time'):
        return {'type': 'observation.available', 'payload': {'observerRef': 'fixture-observer', 'values': {'state': 1.0}, 'units': {'state': unit}},
                'sourceTime': {'tick': tick, 'clockDomain': clock, 'resolution': 'microsecond', 'uncertainty': 0}}
    records = [observation('0'), observation('5'), {'type': 'evidence.updated', 'payload': {'coverage': {'clockDomain': 'simulation-time', 'startTick': '0', 'endTick': '10', 'complete': True}}}]
    healthy = recorded_input(run, snapshot, records, 'valid', snapshot_integrity='valid')
    assert compare_runs(healthy, healthy)['compatible'] is True
    mixed = deepcopy(records); mixed[1]['sourceTime']['clockDomain'] = 'other-clock'
    out = recorded_input(run, snapshot, mixed, 'valid', snapshot_integrity='valid')
    assert out['observations'][0]['metadataConsistent'] is False
    assert compare_runs(healthy, out)['compatible'] is False
    gapped = [*records, {'type': 'observation.gap', 'payload': {'reason': 'labelled fixture gap'}}]
    out = recorded_input(run, snapshot, gapped, 'valid', snapshot_integrity='valid')
    assert out['observations'][0]['coverage']['gapCount'] == 1
    assert compare_runs(healthy, out)['compatible'] is False

@pytest.mark.parametrize('mode,expected', [('unsupported-unit', 'QUANTITY_INCOMPATIBLE'), ('clock', 'CLOCK_INCOMPATIBLE'), ('gap', 'OBSERVATION_COVERAGE_INCOMPLETE'), ('missing-coverage', 'OBSERVATION_WINDOW_UNAVAILABLE')])
def test_public_api_handoff_quality_fixtures_cannot_upgrade_actual_evidence(client, monkeypatch, mode, expected):
    """Post-reader fault fixtures, not actual canonical tampering or provider admission."""
    from test_advanced_api import prepared, start, wait_run
    import xverse_studio.api.advanced as advanced
    project, pf = prepared(client); left = start(client, pf, key='quality-left')
    assert wait_run(client, left)['state'] == 'stopped'
    pf = client.post(f"/api/v1/projects/{project['projectId']}/preflights", json={'revision': project['revision']}).json()
    right = start(client, pf, key='quality-right'); assert wait_run(client, right)['state'] == 'stopped'
    store = client.app.state.studio.store
    paths = [store.root/'runs'/rid/'evidence.jsonl' for rid in (left, right)]
    protected = {p: hashlib.sha256(p.read_bytes()).hexdigest() for p in paths}
    payload = {'leftRunId': left, 'rightRunId': right}
    healthy = client.post('/api/v1/comparisons', json=payload)
    assert healthy.status_code == 200 and healthy.json()['compatible'] is True
    original = advanced.read_journal
    def handoff(root, state):
        records, integrity = original(root, state)
        if state['runId'] != right: return records, integrity
        records = deepcopy(records)
        for record in records:
            if record['type'] == 'observation.available':
                if mode == 'unsupported-unit': record['payload']['units'] = {k: 'fixture-unsupported-unit' for k in record['payload']['values']}
                if mode == 'clock': record['sourceTime']['clockDomain'] = 'fixture-other-clock'
            if mode == 'clock' and isinstance(record['payload'].get('coverage'), dict): record['payload']['coverage']['clockDomain'] = 'fixture-other-clock'
        if mode == 'gap': records.append({'type': 'observation.gap', 'payload': {'reason': 'labelled post-reader gap fixture'}})
        if mode == 'missing-coverage': records = [r for r in records if not isinstance(r['payload'].get('coverage'), dict)]
        return records, integrity
    monkeypatch.setattr(advanced, 'read_journal', handoff)
    response = client.post('/api/v1/comparisons', json=payload)
    assert response.status_code == 200, response.text
    out = response.json(); assert out['compatible'] is False and expected in out['diagnostics']
    assert out['left']['state'] == out['right']['state'] == 'stopped'
    assert out['left']['cleanup'] == out['right']['cleanup'] == 'complete'
    assert all(hashlib.sha256(p.read_bytes()).hexdigest() == h for p,h in protected.items())
    assert len(client.get('/api/v1/runs').json()['items']) == 2
    assert client.get('/api/v1/projects/'+project['projectId']).json() == project


def test_actual_running_evidence_is_not_upgraded_to_a_complete_comparison(client):
    import time
    from test_advanced_api import prepared, start, wait_run
    project, _ = prepared(client)
    edited = client.post(f"/api/v1/projects/{project['projectId']}/edits", headers={'If-Match': str(project['revision'])},
                         json={'baseRevision': project['revision'], 'edits': [{'kind': 'set_parameter', 'parameterId': 'duration', 'value': 3.0}]})
    assert edited.status_code == 200
    project = edited.json(); pf = client.post(f"/api/v1/projects/{project['projectId']}/preflights", json={'revision': project['revision']}).json()
    assert pf['status'] == 'ready'
    rid = start(client, pf, key='running-comparison')
    try:
        deadline = time.monotonic()+20
        while time.monotonic() < deadline:
            current = client.get('/api/v1/runs/'+rid).json()
            if current['state'] == 'running': break
            time.sleep(.05)
        assert current['state'] == 'running'
        response = client.post('/api/v1/comparisons', json={'leftRunId': rid, 'rightRunId': rid})
        assert response.status_code == 200, response.text
        out = response.json(); assert out['compatible'] is False
        assert 'EVIDENCE_INCOMPLETE' in out['diagnostics']
        assert all(f['compatible'] is False for f in out['fields'])
        assert len(client.get('/api/v1/runs').json()['items']) == 1
    finally:
        assert wait_run(client, rid)['state'] == 'stopped'
