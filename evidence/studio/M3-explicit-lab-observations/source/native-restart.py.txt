from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import os
import selectors
import signal
import sqlite3
import subprocess
import sys
import threading
import time
import httpx

assert sys.platform == 'linux'
root = Path.cwd().resolve()
base = root / '.studio-local/lab-probe-foundation'
out = base / 'native-restart'
out.mkdir(mode=0o700)
data = out / 'Owned observation Ω'
host = base / 'native-host.py'
children = []
events = []

def identity(child, argv):
    proc = Path('/proc') / str(child.pid)
    assert child.poll() is None and proc.stat().st_uid == os.getuid()
    fields = (proc / 'stat').read_text().split(') ', 1)[1].split()
    assert int(fields[1]) == os.getpid()
    assert (proc / 'cmdline').read_bytes().split(b'\0')[:-1] == [os.fsencode(arg) for arg in argv]
    return dict(pid=child.pid, parent=int(fields[1]), birthTicks=fields[19], argv=argv)

def start(mode):
    argv = [sys.executable, '-I', str(host), '--data', str(data), '--mode', mode]
    with (out / (mode + '-stderr.private.log')).open('wb') as log:
        child = subprocess.Popen(argv, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE, stderr=log, cwd=root)
    original = identity(child, argv)
    children.append((child, argv, original))
    with selectors.DefaultSelector() as selector:
        selector.register(child.stdout, selectors.EVENT_READ)
        assert selector.select(15), 'Owned host startup announcement timed out'
        line = child.stdout.readline(4097)
    assert len(line) <= 4096 and line.endswith(b'\n')
    origin = json.loads(line)['origin']
    until = time.monotonic() + 15
    while time.monotonic() < until:
        try:
            if httpx.get(origin + '/api/v1/session', timeout=1).status_code == 401:
                break
        except httpx.HTTPError:
            pass
        time.sleep(0.05)
    else:
        raise AssertionError('Owned host readiness timed out')
    client = httpx.Client(base_url=origin, headers={'Origin': origin}, timeout=10)
    bootstrap = json.loads((data / 'bootstrap.json').read_text())
    response = client.post('/api/v1/session', json={'bootstrapToken': bootstrap['bootstrapToken']})
    assert response.status_code == 200
    client.headers['X-CSRF-Token'] = response.json()['csrfToken']
    return child, argv, original, client

def stop(child, argv, original):
    assert identity(child, argv) == original
    child.terminate()
    child.wait(10)
    events.append(dict(kind='owned-normal-stop', process=original, exitCode=child.returncode))

try:
    child, argv, original, client = start('blocked')
    option = client.get('/api/v1/assets/native-observation/probe-options').json()
    assert option['allowed']
    payload = dict(expectedBindingDigest=option['installation']['bindingDigest'])
    response = {}
    def submit():
        try:
            value = client.post('/api/v1/assets/native-observation/probes',
                                headers={'Idempotency-Key': 'original-owned-probe'}, json=payload)
            response.update(status=value.status_code, body=value.json())
        except httpx.HTTPError as error:
            response['transportFailure'] = type(error).__name__
    request = threading.Thread(target=submit)
    request.start()
    marker = data / 'owned-callback-paused.json'
    until = time.monotonic() + 4
    while time.monotonic() < until and not marker.exists():
        assert child.poll() is None
        time.sleep(0.01)
    assert marker.exists(), 'Provider callback did not reach the owned fault point'
    paused = json.loads(marker.read_text())
    assert time.time() < paused['deadline']
    with sqlite3.connect('file:' + str(data / 'studio.sqlite') + '?mode=ro', uri=True) as db:
        before = json.loads(db.execute("SELECT body FROM records WHERE kind='operation' AND id=?", (paused['operationId'],)).fetchone()[0])
        mapped = json.loads(db.execute('SELECT result FROM idempotency WHERE scope=? AND key=?',
            ('local-operator:POST /assets/native-observation/probes', 'original-owned-probe')).fetchone()[0])
    assert before['status'] == 'delegated' and before['result'] is None
    assert mapped['operationId'] == before['operationId']
    assert json.loads((data / 'fixture-contact-count.json').read_text()) == 1
    assert identity(child, argv) == original
    child.kill()
    child.wait(10)
    assert child.returncode == -signal.SIGKILL
    request.join(10)
    assert not request.is_alive() and 'transportFailure' in response
    client.close()
    events.append(dict(kind='owned-sigkill-after-fixture-contact-before-ack', process=original,
                       committedOperation=before, committedKeyMapping=mapped, response=response, exitCode=child.returncode))
    child, argv, original, client = start('normal')
    observed = client.get('/api/v1/operations/' + before['operationId']).json()
    assert observed['status'] == 'unknown' and observed['code'] == 'PROBE_INTERRUPTED'
    assert observed['result'] is None and observed['bindingDigest'] == before['bindingDigest']
    assert client.post('/api/v1/assets/native-observation/probes', headers={'Idempotency-Key': 'original-owned-probe'}, json=payload).json() == observed
    assert json.loads((data / 'fixture-contact-count.json').read_text()) == 1
    newer = client.post('/api/v1/assets/native-observation/probes', headers={'Idempotency-Key': 'new-explicit-owned-probe'}, json=payload).json()
    assert newer['status'] == 'completed' and newer['result']['status'] == 'observed'
    assert newer['operationId'] != before['operationId']
    assert json.loads((data / 'fixture-contact-count.json').read_text()) == 2
    assert client.get('/api/v1/operations/' + before['operationId']).json() == observed
    events.append(dict(kind='restart-never-replays-original-new-explicit-request-is-distinct', original=observed, new=newer, fixtureContacts=2))
    client.close()
    stop(child, argv, original)
    child, argv, original, client = start('uninstalled')
    assert client.get('/api/v1/capabilities').json()['labProbes']['status'] == 'unavailable'
    assert client.get('/api/v1/operations/' + before['operationId']).json() == observed
    assert client.post('/api/v1/assets/native-observation/probes', headers={'Idempotency-Key': 'original-owned-probe'}, json=payload).json() == observed
    assert json.loads((data / 'fixture-contact-count.json').read_text()) == 2
    client.close()
    stop(child, argv, original)
    result = dict(observedAt=datetime.now(timezone.utc).isoformat(), outcome='passed', events=events,
                  faultPoint='After an owned fixture JSON contact counter increment, before observation acknowledgement; real exact-owner Linux SIGKILL/restart.',
                  fixtureScope='Durable JSON counter is a fixture witness, not a surviving external provider or physical-device action.',
                  allChildrenTerminal=all(p.poll() is not None for p, a, i in children),
                  hostSourceSha256=hashlib.sha256(host.read_bytes()).hexdigest(),
                  controllerSourceSha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest())
    (out / 'verification.private.json').write_text(json.dumps(result, indent=2) + '\n')
    print(json.dumps({k:v for k,v in result.items() if k != 'events'}), flush=True)
finally:
    for child, argv, original in children:
        if child.poll() is None:
            assert identity(child, argv) == original
            child.terminate()
            child.wait(10)
