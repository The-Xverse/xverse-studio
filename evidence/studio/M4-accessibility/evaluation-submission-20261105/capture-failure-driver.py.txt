import argparse
import hashlib
import json
import os
from pathlib import Path
import select
import signal
import subprocess
import sys
import time

import psutil

parser = argparse.ArgumentParser()
parser.add_argument('phase')
parser.add_argument('--ui', required=True, type=Path)
parser.add_argument('--suite', default='evaluation-keyboard')
parser.add_argument('--grep')
a = parser.parse_args()
root = Path.cwd()
out = root / '.studio-local/t097-evaluation-keyboard-20261105' / a.phase
out.mkdir(exist_ok=False)
sha = lambda data: hashlib.sha256(data).hexdigest()
paths = ['apps','services','packages','tests','config','scripts','package.json','package-lock.json','pyproject.toml','uv.lock']
names = sorted(set(n for args in [('ls-files','-z','--',*paths),('ls-files','-z','--others','--exclude-standard','--',*paths)] for n in subprocess.check_output(['git',*args], text=True).split('\0') if n))
def inputs(): return {n:sha(Path(n).read_bytes()) for n in names}
before = inputs()
ui = a.ui.resolve()
def assets(): return {p.relative_to(ui).as_posix():sha(p.read_bytes()) for p in ui.rglob('*') if p.is_file()}
ui_before = assets()
(out/'source-inputs.json').write_text(json.dumps(before,indent=2)+'\n')
(out/'ui-inputs.json').write_text(json.dumps(ui_before,indent=2)+'\n')
prefix = root.parents[3]/'.xverse-studio-cache'
argv = [sys.executable,'-I','tests/live/autoverse_playback_browser.py','--recording',str(root/'.studio-local/t088-assessment-mapping/native-newenv-5/demo'),'--static-dir',str(ui),'--output',str(out/'native'),'--node',str(prefix/'node-v24.22.1-linux-x64/bin/node'),'--playwright-cli','[MAIN_CHECKOUT]/node_modules/@playwright/test/cli.js','--browser-cache',str(prefix/'playwright'),'--isolate-browser-devices','--suite',a.suite]
if a.grep: argv += ['--case-filter', a.grep]
if a.suite == 'evaluation-keyboard': argv += ['--receipt', str(root/'.studio-local/t097-evaluation-keyboard-20261105/installation-current.private.json')]
argv = [sys.executable, '-I', str(root/'.studio-local/t097-evaluation-keyboard-20261105/enter-native.py'), *argv]
(out/'command.private.json').write_text(json.dumps({'argv':argv},indent=2)+'\n')
records, gaps, gpu = {}, 0, []
started = time.monotonic()
with (out/'driver.log').open('xb') as log:
    child = subprocess.Popen(argv,env=dict(os.environ,PYTHONPATH=''),stdin=subprocess.DEVNULL,stdout=log,stderr=subprocess.STDOUT)
    owner = psutil.Process(child.pid)
    birth = owner.create_time()
    handle = os.pidfd_open(child.pid)
    try:
        while child.poll() is None:
            assert time.monotonic()-started < 425
            for p in owner.children(recursive=True):
                fd = None
                try:
                    key = (p.pid,p.create_time())
                    if key in records: continue
                    assert len(records)<512 and p.uids().real==os.getuid()
                    argv_child = p.cmdline()
                    fd = os.pidfd_open(p.pid)
                    if p.create_time()!=key[1]: raise psutil.NoSuchProcess(p.pid)
                    for entry in list((Path('/proc')/str(p.pid)/'fd').iterdir())[:1024]:
                        try:
                            target=os.readlink(entry)
                            if target.startswith(('/dev/nvidia','/dev/dri/')): gpu.append({'pid':p.pid,'path':target})
                        except OSError: pass
                    records[key]={'pid':p.pid,'birth':key[1],'argv':argv_child,'pidfd':fd}
                    fd=None
                except (psutil.NoSuchProcess,psutil.ZombieProcess,FileNotFoundError,ProcessLookupError): gaps+=1
                finally:
                    if fd is not None: os.close(fd)
            assert not gpu
            time.sleep(.1)
        child.wait(timeout=10)
    finally:
        if child.poll() is None:
            assert owner.create_time()==birth and owner.uids().real==os.getuid()
            signal.pidfd_send_signal(handle,signal.SIGTERM,None,0)
            child.wait(timeout=35)
        os.close(handle)
deadline=time.monotonic()+10
while True:
    pending=[v for v in records.values() if not select.select([v['pidfd']],[],[],0)[0]]
    if not pending: break
    assert time.monotonic()<deadline
    time.sleep(.1)
for v in records.values(): os.close(v.pop('pidfd'))
(out/'observed-processes.private.json').write_text(json.dumps(list(records.values()),indent=2)+'\n')
proof=json.loads((out/'native/browser-device-proof.private.json').read_text())
native=json.loads((out/'native/verification.json').read_text())
report=json.loads((out/'native/browser.json').read_text())
assert proof['namespace']!=proof['hostNamespace'] and proof['uid']==os.getuid() and proof['displaysUnset']
assert not any(n=='dri' or n.startswith('nvidia') for n in proof['devices'])
assert native['originalBrowserExited'] and native['originalRootReleased']
assert native['originalRecordingUnchanged'] and native['copiedOriginalCanonicalUnchanged'] and native['after']==native['expectedExecutionCounts']
assert inputs()==before and assets()==ui_before
result={'sourceCandidate':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),'sourceInputs':len(before),'sourceInputsUnchanged':True,'uiInputsUnchanged':True,'childExit':child.returncode,'stats':report['stats'],'observedProcessesExited':len(records),'failedProcessCaptureAttempts':gaps,'allShortLivedChildrenObserved':False,'gpuFdsObserved':gpu,'deviceProof':proof,'native':native}
(out/'verification.private.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({'phase':a.phase,'exit':child.returncode,'stats':report['stats'],'observedProcessesExited':len(records),'gpuFdsObserved':len(gpu),'canonicalUnchangedAndLedgerAccounted':True,'sourceInputsUnchanged':True}),flush=True)
raise SystemExit(child.returncode)
