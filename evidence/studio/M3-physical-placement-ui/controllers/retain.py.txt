def read(p):
    return json.loads(p.read_text())

def clean_text(v):
    for a, b in [(str(root.parent / '.xverse-studio-cache'), '<loop-cache>'),
                 (str(root), '<repo>'), ('<repo>', '<repo>'),
                 (sys.base_prefix, '<python-install>')]:
        v = v.replace(a, b)
    return v

def clean(v):
    if isinstance(v, str):
        return clean_text(v)
    if isinstance(v, list):
        return [clean(x) for x in v]
    if isinstance(v, dict):
        result = {k: clean(x) for k, x in v.items()}
        if v.get('contentType', '').startswith(('application/json', 'text/')) and 'body' in v:
            try:
                result['body'] = base64.b64encode(clean_text(base64.b64decode(v['body']).decode()).encode()).decode()
            except (ValueError, UnicodeError):
                pass
        return result
    return v

def save(n, v):
    p = out / n
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(clean(v), indent=2) + '\n')

def copy_text(n, source):
    p = out / n
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text('\n'.join(x.rstrip() for x in clean_text(source.read_text()).splitlines()).rstrip() + '\n')

def results(report):
    for suite in report.get('suites', []):
        yield from results(suite)
    for spec in report.get('specs', []):
        for test in spec['tests']:
            for result in test['results']:
                yield spec['title'], result

def attachments(stage, report):
    pairs = []
    for title, result in results(report):
        vp = re.search(r'(\d+x\d+)', title)
        viewport = vp.group(1) if vp else 'unspecified'
        mode = re.sub(r'[^A-Za-z0-9_-]', '-', title.split(': ', 1)[-1].rsplit(' at ', 1)[0])
        for a in result.get('attachments', []):
            if a.get('contentType') == 'application/json' and 'body' in a:
                value = json.loads(base64.b64decode(a['body']))
                save(f'{stage}/journeys/{viewport}/{mode}/{a["name"]}', value)
                if isinstance(value, dict):
                    pairs.extend(value.get('measurements', []))
                    selected = value.get('selectedContrast')
                    pairs.extend(selected if isinstance(selected, list) else [selected] if isinstance(selected, dict) else [])
    return pairs
