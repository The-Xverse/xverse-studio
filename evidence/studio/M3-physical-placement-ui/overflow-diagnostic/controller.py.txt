from pathlib import Path
import hashlib
import json
import os
import subprocess
import sys
import time

root = Path.cwd(); out = root/'.studio-local/physical-placements/overflow-diagnostic'
cache = root.parent/'.xverse-studio-cache'
env = dict(os.environ, PATH=str(cache/'node-v24.22.1-linux-x64/bin')+os.pathsep+os.environ['PATH'],
    TMPDIR=str(cache/'tmp'), TEMP=str(cache/'tmp'), TMP=str(cache/'tmp'),
    UV_CACHE_DIR=str(cache/'uv'), npm_config_cache=str(cache/'npm'),
    PLAYWRIGHT_BROWSERS_PATH=str(cache/'playwright'), PYTEST_DISABLE_PLUGIN_AUTOLOAD='1')
extra = ['docs/architecture/ADR-0018-physical-binding-observations.md','docs/architecture/ADR-0017-owned-domain-presentation.md','docs/architecture/ADR-0013-revision-history-and-recovery.md','docs/architecture/ADR-0015-deployment-resolution.md','.github/workflows/ci.yml','.python-version','docs/implementation/baseline.md',
    'docs/implementation/dependencies.md','docs/implementation/platform-support.md','docs/implementation/sads-dispositions.md',
    'evidence/studio/M3-physical-contracts/environment.json','evidence/studio/M0-linux-baseline/verification.json',
    'evidence/studio/M1-shell-operability/verification.json',
    'evidence/studio/M1-shell-operability/journeys/1024x768/login/native-browser-environment.json',
    'evidence/studio/native-ci/windows-initial-install-failure.json']
def hashes():
    names = subprocess.check_output(['git','ls-files','-z','--','apps','services','scripts','tests','packages','config','pyproject.toml','uv.lock','package.json','package-lock.json'], text=True).split('\0')
    names += subprocess.check_output(['git','ls-files','--others','--exclude-standard','-z','--','apps','services','scripts','tests','packages','config'],text=True).split('\0')
    names += [str(p.relative_to(root)) for p in (root/'apps/studio-web/dist').rglob('*') if p.is_file()]
    return {n:hashlib.sha256((root/n).read_bytes()).hexdigest() for n in sorted(set(names+extra)) if n and (root/n).is_file()}
before = hashes(); (out/'source-before.json').write_text(json.dumps(before,indent=2)+'\n')
start = time.monotonic()
with (out/'browser.log').open('w') as log:
    result = subprocess.run([sys.executable,'-I','scripts/test_browser.py','--grep','physical placements: labelled matched'], env=env, stdout=log, stderr=subprocess.STDOUT)
after = hashes(); (out/'source-after.json').write_text(json.dumps(after,indent=2)+'\n')
summary = {'baseCandidate':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),
    'command':['.venv/bin/python','-I','scripts/test_browser.py','--grep','physical placements: labelled matched'],'exitCode':result.returncode,
    'seconds':round(time.monotonic()-start,3),'sourceUnchanged':before==after}
(out/'summary.json').write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps(summary),flush=True)

raise SystemExit(result.returncode)
