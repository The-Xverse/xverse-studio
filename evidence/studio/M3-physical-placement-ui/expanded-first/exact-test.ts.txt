async function tabTo(page: Page, target: Locator) {
  await expect(target).toBeVisible();
  for (const [key, limit] of [
    ["Shift+Tab", 40],
    ["Tab", 220],
  ] as const) {
    for (let i = 0; i < limit; i++) {
      if (await target.evaluate((n) => document.activeElement === n)) return;
      await page.keyboard.press(key);
    }
  }
  await expect(target).toBeFocused();
}
async function activate(page: Page, target: Locator) {
  await tabTo(page, target);
  await page.keyboard.press("Enter");
}
async function request(
  page: Page,
  route: string,
  method = "GET",
  body?: unknown,
  revision?: number,
) {
  return page.evaluate(
    async ({ route, method, body, revision }) => {
      const { csrfToken } = await fetch("/api/v1/session").then((r) =>
        r.json(),
      );
      const response = await fetch("/api/v1" + route, {
        method,
        headers: {
          "Content-Type": "application/json",
          "X-CSRF-Token": csrfToken,
          ...(revision === undefined ? {} : { "If-Match": `"${revision}"` }),
        },
        body: body === undefined ? undefined : JSON.stringify(body),
      });
      const value = await response.json();
      if (!response.ok)
        throw new Error(
          `Owned placement fixture request failed: ${response.status} ${value.code}`,
        );
      return value;
    },
    { route, method, body, revision },
  );
}

async function hold(page: Page, status: 200 | 503 = 200) {
  let release!: () => void, finished!: () => void, ready = false, departed = false, reads = 0;
  const gate = new Promise<void>(r => release=r), done = new Promise<void>(r => finished=r);
  await page.route(pattern, async route => {
    reads++;
    try {
      const response = status === 200 ? await route.fetch() : undefined;
      if (response) expect(response.status()).toBe(200);
      ready=true;
      await gate;
      if (response) await route.fulfill({response});
      else await route.fulfill({status:503,contentType:"application/json",body:JSON.stringify({code:"PLACEMENT_OBSERVATION_UNAVAILABLE",message:"Labelled placement transport failure"})});
    } catch (error) { if(!departed) throw error; }
    finally { finished(); }
  });
  return { ready:()=>ready, reads:()=>reads, release, done, depart:()=>{departed=true;}, async close(){departed=true;release(); if(reads) await done;await page.unroute(pattern);} };
}

  await page.waitForLoadState("networkidle");
}

async function observeWrites(page: Page, project: any, info: TestInfo) {
  const beforeRuns = await request(page, "/runs"), reads: string[] = [];
  let writes = 0;
  const listener = (r: import("@playwright/test").Request) => {
    if (!["GET", "HEAD"].includes(r.method())) writes++;
    if (new URL(r.url()).pathname.endsWith("/physical-bindings")) reads.push(new URL(r.url()).pathname + new URL(r.url()).search);
  };
  page.on("request", listener);
  return async (details: Record<string, unknown>) => {
    await page.unroute(pattern);
    await page.waitForLoadState("networkidle");
    expect(await request(page, `/projects/${project.projectId}`)).toEqual(project);
    expect(await request(page, "/runs")).toEqual(beforeRuns);
    expect(writes).toBe(0);
    page.off("request", listener);
    await info.attach("placement-journey.json", {
      body: JSON.stringify({ projectId: project.projectId, revision: project.revision, reads, mutativeRequestsAfterSetup: writes,
        canonicalProjectAndRunsUnchanged: true, ...details,
        scope: "Read-only native browser inspection; actual API results and labelled transport fixtures are distinguished. No physical provider installed, reservation, device operation or Start." }),
      contentType: "application/json",
    });
  };
}

async function bounds(target: Locator, viewport: {width:number;height:number}) {
  await expect(target).toBeFocused();
  const geometry = await target.evaluate(n => {const box=n.getBoundingClientRect(), style=getComputedStyle(n);return {box:box.toJSON(),outline:style.outlineStyle,outlineWidth:parseFloat(style.outlineWidth)};});
  expect(geometry.box.left).toBeGreaterThanOrEqual(0);
  expect(geometry.box.top).toBeGreaterThanOrEqual(0);
  expect(geometry.box.right).toBeLessThanOrEqual(viewport.width);
  expect(geometry.box.bottom).toBeLessThanOrEqual(viewport.height);
  expect(geometry.outline).not.toBe("none");
  expect(geometry.outlineWidth).toBeGreaterThan(0);
  return geometry;
}

async function paint(target: Locator, label: string) {
  const observed = await renderedContrast(target, label);
  expect(observed.supported).toBe(true);
  if (observed.supported) expect(observed.ratio).toBeGreaterThanOrEqual(4.5);
  return observed;
}
