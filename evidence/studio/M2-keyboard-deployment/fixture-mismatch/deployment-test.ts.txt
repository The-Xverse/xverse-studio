async function tabTo(page: Page, target: Locator) {
  await expect(target).toBeVisible();
  for (let i = 0; i < 220; i++) {
    if (await target.evaluate((node) => document.activeElement === node))
      return;
    await page.keyboard.press("Tab");
  }
  throw Error("Deployment control is unreachable through native Tab order");
}
async function activate(page: Page, target: Locator) {
  await tabTo(page, target);
  await page.keyboard.press("Enter");
}
async function type(page: Page, target: Locator, value: string) {
  await tabTo(page, target);
  await page.keyboard.press("ControlOrMeta+A");
  await page.keyboard.insertText(value);
}
async function choose(page: Page, target: Locator, value: string) {
  const enabled = await target
    .locator("option")
    .evaluateAll((nodes) =>
      nodes
        .filter((n) => !(n as HTMLOptionElement).disabled)
        .map((n) => (n as HTMLOptionElement).value),
    );
  const index = enabled.indexOf(value);
  expect(index).toBeGreaterThanOrEqual(0);
  await tabTo(page, target);
  await page.keyboard.press("Home");
  for (let i = 0; i < index; i++) await page.keyboard.press("ArrowDown");
  await expect(target).toHaveValue(value);
}
function document(project: any, suffix: string): any {
  return Object.values(project.documents).find((d: any) =>
    d.path.endsWith("/" + suffix),
  );
}
async function project(page: Page, name: string) {
  let result = await page.evaluate(async (name) => {
    const { csrfToken } = await fetch("/api/v1/session").then((r) => r.json());
    const r = await fetch("/api/v1/projects", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "X-CSRF-Token": csrfToken,
      },
      body: JSON.stringify({ name }),
    });
    if (!r.ok) throw Error("Owned deployment fixture creation failed");
    return r.json();
  }, name);
  const replacements = [
    [
      "system.xdl.yaml",
      "  defaultTimeDomainId: simulation-time\n",
      "  - id: wall-time\n    clockClass: wall\n    epoch: unix\n    rate: '1'\n    monotonic: false\n  defaultTimeDomainId: simulation-time\n",
    ],
    [
      "deployment.xdl.yaml",
      "  targets:\n  - id: reference-target\n",
      "  resources:\n  - id: bench-cpu\n    resourceClass: cpu\n    capacity:\n      value: 2\n      unit: core\n    allocationPolicy: exclusive\n  targets:\n  - id: reference-target\n    offeredResourceIds:\n    - bench-cpu\n",
    ],
  ];
  for (const [suffix, from, to] of replacements) {
    const [id, d]: any = Object.entries(result.documents).find(([, d]: any) =>
      d.path.endsWith("/" + suffix),
    );
    expect(d.content).toContain(from);
    result = await page.evaluate(
      async ({ projectId, revision, id, content }) => {
        const { csrfToken } = await fetch("/api/v1/session").then((r) =>
          r.json(),
        );
        const r = await fetch(`/api/v1/projects/${projectId}/documents/${id}`, {
          method: "PUT",
          headers: {
            "Content-Type": "application/json",
            "X-CSRF-Token": csrfToken,
            "If-Match": String(revision),
          },
          body: JSON.stringify({ content }),
        });
        if (!r.ok) throw Error("Owned deployment fixture preparation failed");
        return r.json();
      },
      {
        projectId: result.projectId,
        revision: result.revision,
        id,
        content: d.content.replace(from, to),
      },
    );
  }
  await page.evaluate(
    (id) => localStorage.setItem("studio.projectId", id),
    result.projectId,
  );
  await page.reload();
  await expect(page.getByRole("heading", { name, exact: true })).toBeVisible();
  await activate(page, page.getByRole("tab", { name: /Deployment/ }));
  await expect(
    page.getByLabel("Deployment variant", { exact: true }),
  ).not.toHaveValue("");
  await expect(
    page.getByLabel("Associated Scenario", { exact: true }),
  ).not.toHaveValue("");
  return result;
}
async function read(page: Page, id: string) {
  return page.request.get(`/api/v1/projects/${id}`).then((r) => r.json());
}
async function runIds(page: Page, id: string) {
  return (await page.request.get("/api/v1/runs").then((r) => r.json())).items
    .filter((r: any) => r.projectId === id)
    .map((r: any) => r.runId)
    .sort();
}
async function mapping(page: Page) {
  const time = page.getByRole("region", { name: "Time mappings", exact: true });
  await type(page, time.getByLabel("Mapping identifier"), "wall-to-sim");
  await choose(page, time.getByLabel("Source clock"), "wall-time");
  await choose(page, time.getByLabel("Target clock"), "simulation-time");
  await type(page, time.getByLabel("Mapping description"), "offset-at-start");
  await type(page, time.getByLabel("Tolerance", { exact: true }), "0.01");
  await type(page, time.getByLabel("Tolerance unit"), "s");
  return time;
}
async function resources(page: Page) {
  await choose(page, page.getByLabel("Binding for resources"), "plant-binding");
  const checkbox = page.getByRole("checkbox", { name: /^bench-cpu/ });
  await tabTo(page, checkbox);
  await page.keyboard.press("Space");
  await expect(checkbox).toBeChecked();
}
async function focusGeometry(target: Locator) {
  return target.evaluate((node) => {
    const rect = node.getBoundingClientRect(),
      style = getComputedStyle(node);
    return {
      focused: document.activeElement === node,
      activeTag: document.activeElement?.tagName,
      rect: rect.toJSON(),
      viewport: { width: innerWidth, height: innerHeight },
      outline: style.outlineStyle,
      outlineWidth: parseFloat(style.outlineWidth),
    };
  });
}
async function hold(page: Page, pattern: string, method: string) {
  let release!: () => void,
    ready = false,
    actual: any = null,
    finished: Promise<void> | undefined;
  const held = new Promise<void>((resolve) => {
    release = resolve;
  });
  await page.route(pattern, async (route) => {
    if (route.request().method() !== method) {
      await route.continue();
      return;
    }
    let finish!: () => void;
    finished = new Promise<void>((resolve) => {
      finish = resolve;
    });
    try {
      const response = await route.fetch();
      actual = { status: response.status(), body: await response.json() };
      ready = true;
      await held;
      await route.fulfill({ response });
    } finally {
      finish();
    }
  });
  return {
    async ready() {
      await expect.poll(() => ready).toBe(true);
      return actual;
    },
    async release() {
      release();
      await finished;
    },
    async close() {
      release();
      await finished;
      await page.unroute(pattern);
    },
  };
}
