import {test,expect,type Locator,type Page} from '@playwright/test';
import {connectStudio} from '../../tests/e2e/connect-studio';
import {traceApi} from '../../tests/e2e/trace-claims-flow';

async function tabTo(page:Page,target:Locator){await expect(target).toBeVisible();for(let i=0;i<160;i++){if(await target.evaluate(node=>document.activeElement===node))return;await page.keyboard.press('Tab');}throw Error('Owned parameter control was not reachable by native Tab.');}
for(const width of [1024,1440]) test(`parameter presentation: real regrouping preserves Apply focus at ${width}`,async({page},info)=>{
 await page.setViewportSize({width,height:width===1024?768:900});await connectStudio(page,info);
 const project=(await traceApi(page,'/projects','POST',{name:'Late presentation keyboard control '+width})).body;
 await page.evaluate(id=>localStorage.setItem('studio.projectId',id),project.projectId);
 let release!:()=>void,ready!:()=>void,finish!:()=>void;const held=new Promise<void>(r=>release=r),actual=new Promise<void>(r=>ready=r),delivered=new Promise<void>(r=>finish=r);let presentation:any;
 const pattern=`**/api/v1/projects/${project.projectId}/presentation`;
 await page.route(pattern,async route=>{const response=await route.fetch();expect(response.status()).toBe(200);const text=await response.text();presentation=JSON.parse(text);const headers={...response.headers()};delete headers['content-length'];ready();await held;try{await route.fulfill({status:200,headers,body:text});}finally{finish();}});
 const edits:any[]=[];page.on('request',request=>{if(request.method()==='POST'&&request.url().endsWith(`/projects/${project.projectId}/edits`))edits.push(request.postDataJSON());});
 try{
  await page.reload();await actual;await expect(page.getByRole('heading',{name:project.name,exact:true})).toBeVisible();
  const field=page.getByLabel('Duration'),button=field.locator('xpath=ancestor::form').getByRole('button',{name:'Apply',exact:true});
  await tabTo(page,field);await page.keyboard.press('ControlOrMeta+A');await page.keyboard.insertText('7');await expect(field).toHaveValue('7');await tabTo(page,button);await expect(button).toBeFocused();
  expect(presentation.formGroups.length).toBeGreaterThan(0);release();await delivered;await expect(page.getByRole('group',{name:'Run horizon',exact:true})).toBeVisible();
  const beforeEnter=await button.evaluate(node=>({focused:document.activeElement===node,activeTag:document.activeElement?.tagName,value:(node.closest('form')?.querySelector('input') as HTMLInputElement)?.value}));
  await info.attach('native-presentation-focus',{contentType:'application/json',body:JSON.stringify({scope:'Actual owned presentation HTTP200 held; no DTO mutation. Native Tab reached Apply before actual metadata regrouped its form.',width,projectId:project.projectId,beforeEnter,edits})});
  await expect(button).toBeFocused();await page.keyboard.press('Enter');await expect(page.locator('.alerts')).toContainText('Draft saved');
  expect(edits).toHaveLength(1);expect(edits[0].edits).toContainEqual(expect.objectContaining({kind:'set_parameter',parameterId:'duration',value:7}));
  const saved=(await traceApi(page,'/projects/'+project.projectId)).body;expect(saved.revision).toBe(project.revision+1);await expect(field).toHaveValue('7');
 }finally{release();await page.unroute(pattern);}
});
