from pathlib import Path
import os,sys,json,uuid,subprocess,socket,time,hashlib
import httpx

root=Path.cwd();base=root/'.studio-local/rejected-rerun-context';out=base/(sys.argv[1] if len(sys.argv)>1 else 'native-red');out.mkdir(exist_ok=False)
uiroot=Path(sys.argv[2]).resolve() if len(sys.argv)>2 else root.resolve()
paths=subprocess.check_output(['git','ls-files','--cached','--others','--exclude-standard'],text=True).splitlines()
paths += [str(p.relative_to(root)) for p in (root/'apps/studio-web/dist').rglob('*') if p.is_file()]
selected={n:hashlib.sha256((root/n).read_bytes()).hexdigest() for n in sorted(set(paths)) if (root/n).is_file() and (n.startswith(('apps/','services/','scripts/','tests/','packages/','config/')) or n in ['pyproject.toml','uv.lock','package.json','package-lock.json','docs/user/run-comparison.md','docs/architecture/ADR-0019-scenario-evidence-timeline.md','docs/architecture/ADR-0020-campaign-workflow.md','docs/architecture/ADR-0021-recorded-comparison.md','docs/architecture/ADR-0022-owned-worker-state-handoff.md'])}
def inputs():return {n:hashlib.sha256((root/n).read_bytes()).hexdigest() for n in selected}
assert inputs()==selected
(out/'source-before.json').write_text(json.dumps(selected,indent=2)+'\n')
data=base/('owned-native-data-'+uuid.uuid4().hex)
cache=root.parent/'.xverse-studio-cache'
env=dict(os.environ,PATH=str(cache/'node-v24.22.1-linux-x64/bin')+os.pathsep+os.environ['PATH'],
         PLAYWRIGHT_BROWSERS_PATH=str(cache/'playwright'),TMPDIR=str(cache/'tmp'),TEMP=str(cache/'tmp'),TMP=str(cache/'tmp'))
assert (uiroot/'apps/studio-web/dist/index.html').is_file(), 'A successful isolated build is required before a browser probe'
with socket.socket() as sock:sock.bind(('127.0.0.1',0));port=sock.getsockname()[1]
url=f'http://127.0.0.1:{port}'
command=[sys.executable,'-I','-m','xverse_studio.host.launcher','--data-dir',str(data),'--port',str(port),'--static-dir',str(uiroot/'apps/studio-web/dist')]
process=subprocess.Popen(command,cwd=root,env=env,stdin=subprocess.DEVNULL,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
ui_before={str(p.relative_to(uiroot)):hashlib.sha256(p.read_bytes()).hexdigest() for p in (uiroot/'apps/studio-web/dist').rglob('*') if p.is_file()}
source='apps/studio-web/src/features/results/ResultsView.tsx'
ui_before[source]=hashlib.sha256((uiroot/source).read_bytes()).hexdigest()
(out/'ui-source-before.json').write_text(json.dumps(ui_before,indent=2)+'\n')
test_before={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in base.iterdir() if p.name.endswith(('.spec.ts','.config.ts'))}
(out/'test-source-before.json').write_text(json.dumps(test_before,indent=2)+'\n')
t=time.monotonic();result=None
try:
    deadline=time.monotonic()+20
    while time.monotonic()<deadline:
        try:
            if httpx.get(url+'/api/v1/session',timeout=1).status_code==401:break
        except httpx.HTTPError:pass
        time.sleep(.1)
    else:raise RuntimeError('Owned probe backend did not start')
    env.update(STUDIO_BASE_URL=url,STUDIO_BROWSER_DATA=str(data),STUDIO_BROWSER_REPORT=str(out/'browser-report.json'),STUDIO_BROWSER_OUTPUT=str(out/'browser-output'))
    command=[str(cache/'node-v24.22.1-linux-x64/bin/node'),str(root/'node_modules/@playwright/test/cli.js'),'test','-c',str(base/'playwright.config.ts'),*(['--grep',sys.argv[3]] if len(sys.argv)>3 else [])]
    with (out/'browser.log').open('w') as log:result=subprocess.run(command,cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT)
finally:
    process.terminate()
    try:process.wait(timeout=15)
    except subprocess.TimeoutExpired:process.kill();process.wait(timeout=5)
    after=inputs();(out/'source-after.json').write_text(json.dumps(after,indent=2)+'\n')
    ui_after={n:hashlib.sha256((uiroot/n).read_bytes()).hexdigest() for n in ui_before}
    (out/'ui-source-after.json').write_text(json.dumps(ui_after,indent=2)+'\n')
    assert ui_before==ui_after
    test_after={n:hashlib.sha256((base/n).read_bytes()).hexdigest() for n in test_before}
    assert test_before==test_after
    (out/'test-source-after.json').write_text(json.dumps(test_after,indent=2)+'\n')
    summary={'uiRoot':str(uiroot),'uiSourceUnchanged':ui_before==ui_after,'command':command,'exitCode':result.returncode if result is not None else None,'seconds':round(time.monotonic()-t,3),'selectedSourceUnchanged':selected==after,'selectedInputs':len(selected),'scope':'Actual owned capacity-rejected rerun recovery probe; selected source bytes unchanged.'}
    (out/'summary.json').write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps(summary),flush=True)
if result is None:raise SystemExit(1)
raise SystemExit(result.returncode or int(selected!=after))
