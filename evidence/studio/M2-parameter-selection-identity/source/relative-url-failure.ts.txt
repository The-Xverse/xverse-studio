import { test, expect, openLab } from "../../../tests/e2e/lab-fixture";
import type { Page, Locator } from "@playwright/test";
async function tabTo(page: Page, target: Locator) {
  await expect(target).toBeVisible();
  for (let n=0;n<200;n++) { if(await target.evaluate(el=>el===document.activeElement)) return; await page.keyboard.press("Tab"); }
  throw Error("Owned parameter control is unreachable");
}
for (const viewport of [{width:1024,height:768},{width:1440,height:900}]) {
 test.describe(`labelled held actual presentation at ${viewport.width}`,()=>{
  test.use({viewport});
  test("metadata arrival preserves native select-all and unsaved value",async({page,labHost},info)=>{
   await openLab(page,labHost);
   const project=await page.evaluate(async()=>{
    const {csrfToken}=await fetch("/api/v1/session").then(r=>r.json());
    const headers={"Content-Type":"application/json","X-CSRF-Token":csrfToken};
    const created=await fetch("/api/v1/projects",{method:"POST",headers,body:JSON.stringify({name:"Owned held presentation selection"})});
    if(created.status!==201)throw Error("Owned creation failed "+created.status);
    const p=await created.json();
    const edited=await fetch(`/api/v1/projects/${p.projectId}/edits`,{method:"POST",headers:{...headers,"If-Match":String(p.revision)},body:JSON.stringify({baseRevision:p.revision,edits:[{kind:"set_parameter",parameterId:"duration",value:6}]})});
    if(edited.status!==200)throw Error("Owned preparation failed "+edited.status);
    const result=await edited.json();localStorage.setItem("studio.projectId",result.projectId);return result;
   });
   let release!:()=>void,entered!:()=>void,settled!:()=>void;
   const held=new Promise<void>(r=>release=r),ready=new Promise<void>(r=>entered=r),done=new Promise<void>(r=>settled=r);
   const pattern=`**/api/v1/projects/${project.projectId}/presentation`;
   await page.route(pattern,async route=>{
    const response=await route.fetch(),body=await response.text();expect(response.status()).toBe(200);
    expect(JSON.parse(body).formGroups.length).toBeGreaterThan(0);entered();await held;
    await route.fulfill({response,body});settled();
   });
   try {
    await page.reload();
    await expect(page.getByRole("heading",{name:project.name,exact:true})).toBeVisible();
    const parameters=page.getByRole("button",{name:"Parameters",exact:true});await tabTo(page,parameters);await page.keyboard.press("Enter");await ready;
    const duration=page.getByLabel("Duration");await expect(duration).toHaveValue("6");await tabTo(page,duration);await page.keyboard.press("ControlOrMeta+A");
    const original=await duration.elementHandle();release();await done;
    await expect(page.getByText("Process control terms",{exact:true})).toBeVisible();
    await page.evaluate(()=>new Promise<void>(resolve=>requestAnimationFrame(()=>requestAnimationFrame(()=>resolve()))));
    await page.keyboard.insertText("7");
    const facts={sameNativeControl:await original!.evaluate(n=>n.isConnected),value:await duration.inputValue(),active:await duration.evaluate(n=>n===document.activeElement),revision:project.revision};
    await info.attach("actual-held-presentation-selection",{body:JSON.stringify(facts),contentType:"application/json"});
    await expect(duration).toHaveValue("7");expect(facts.sameNativeControl).toBe(true);expect(facts.active).toBe(true);
    expect((await (await page.request.get(`/api/v1/projects/${project.projectId}`)).json()).revision).toBe(project.revision);
   } finally {release();await page.unrouteAll({behavior:"wait"});}
  });
 });
}
