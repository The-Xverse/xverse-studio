def read(path):
    return json.loads(path.read_text())

def clean_text(value):
    for original, replacement in [(str(root.parent / '.xverse-studio-cache'), '<loop-cache>'),
                                  (str(root), '<repo>'), ('<repo>', '<repo>'),
                                  (sys.base_prefix, '<python-install>')]:
        value = value.replace(original, replacement)
    return value

def clean(value):
    if isinstance(value, str): return clean_text(value)
    if isinstance(value, list): return [clean(v) for v in value]
    if isinstance(value, dict):
        result = {k: clean(v) for k, v in value.items()}
        if value.get('contentType', '').startswith(('text/', 'application/json')) and 'body' in value:
            try:
                result['body'] = base64.b64encode(clean_text(base64.b64decode(value['body']).decode()).encode()).decode()
            except (ValueError, UnicodeError): pass
        return result
    return value

def save(name, value):
    path = out / name
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(clean(value), indent=2) + '\n')

def copy_text(name, source):
    path = out / name
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text('\n'.join(line.rstrip() for line in clean_text(source.read_text()).splitlines()).rstrip() + '\n')

def results(report):
    for suite in report.get('suites', []): yield from results(suite)
    for spec in report.get('specs', []):
        for test in spec['tests']:
            for result in test['results']: yield spec['title'], result
