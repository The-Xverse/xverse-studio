import { test, expect, type Page } from "@playwright/test";
import { parse, stringify } from "yaml";
import { connectStudio } from "./connect-studio";

async function request(page: Page, path: string, method = "GET", body?: any, revision?: number) {
  return page.evaluate(async ({ path, method, body, revision }) => {
    const { csrfToken } = await fetch("/api/v1/session").then(r => r.json());
    const r = await fetch("/api/v1" + path, { method, headers: {
      "Content-Type": "application/json", "X-CSRF-Token": csrfToken,
      ...(revision ? { "If-Match": String(revision) } : {}),
      ...(method === "POST" ? { "Idempotency-Key": crypto.randomUUID() } : {}),
    }, ...(body === undefined ? {} : { body: JSON.stringify(body) }) });
    const v = await r.json();
    if (!r.ok) throw new Error(`Owned campaign setup ${r.status} ${v.code}`);
    return v;
  }, { path, method, body, revision });
}
async function prepare(page: Page, width: number) {
  let project = await request(page, "/projects", "POST", { name: `Owned campaign history ${width}` });
  const [docId, doc]: any = Object.entries(project.documents).find(([, d]: any) => parse(d.content).kind === "Scenario")!;
  const source = parse(doc.content);
  source.extensions["io.xverse.experiment"].parameters.find((p: any) => p.id === "duration").value = .2;
  project = await request(page, `/projects/${project.projectId}/documents/${docId}`, "PUT", { content: stringify(source), format: "yaml" }, project.revision);
  const pf = await request(page, `/projects/${project.projectId}/preflights`, "POST", { revision: project.revision });
  expect(pf.status).toBe("ready");
  await page.evaluate(({ project, pf }) => {
    localStorage.setItem("studio.projectId", project.projectId);
    localStorage.setItem("studio.selection." + project.projectId, JSON.stringify(pf.selection));
  }, { project, pf });
  await page.reload();
  await page.getByRole("tab", { name: "Experiments", exact: true }).click();
  await page.getByRole("tab", { name: "Campaigns", exact: true }).click();
  return { project, pf };
}
for (const width of [1024, 1440]) {
  test(`campaign workflow: retained native children reopen without scheduling at ${width}`, async ({ page }, info) => {
    await page.setViewportSize({ width, height: width === 1024 ? 768 : 900 });
    await connectStudio(page, info);
    const { project, pf } = await prepare(page, width);
    const made = await request(page, "/campaigns", "POST", { snapshotId: pf.snapshotId, matrix: { gain: [.5, .8] }, budget: { maxRuns: 2, maxTotalSeconds: 1 }, concurrency: 1, stopPolicy: "continue" });
    await expect.poll(async () => (await request(page, `/campaigns/${made.campaignId}`)).status, { timeout: 30000 }).toBe("complete");
    const before = await request(page, `/campaigns/${made.campaignId}`);
    const runsBefore = (await request(page, "/runs")).items.map((r: any) => r.runId);
    const writes: string[] = [];
    page.on("request", r => { if (!["GET", "HEAD", "OPTIONS"].includes(r.method())) writes.push(r.method() + " " + new URL(r.url()).pathname); });
    const refresh = page.getByRole("button", { name: "Refresh campaign history", exact: true });
    await expect(refresh).toBeVisible();
    await refresh.focus(); await page.keyboard.press("Enter");
    const open = page.getByRole("button", { name: `Inspect campaign ${made.campaignId}`, exact: true });
    await expect(open).toBeVisible();
    await open.focus(); await page.keyboard.press("Enter");
    const panel = page.getByRole("region", { name: "Selected campaign", exact: true });
    await expect(panel).toContainText(pf.snapshotId);
    await expect(panel).toContainText(project.projectId);
    for (const c of before.children) {
      await expect(panel).toContainText(c.childKey);
      await expect(panel).toContainText(c.runId);
    }
    await expect(panel).toContainText("complete");
    expect(await request(page, `/campaigns/${made.campaignId}`)).toEqual(before);
    expect((await request(page, "/runs")).items.map((r: any) => r.runId)).toEqual(runsBefore);
    expect(writes).toEqual([]);
    expect(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth + 1)).toBe(true);
    await info.attach("retained-campaign", { contentType: "application/json", body: JSON.stringify({ projectId: project.projectId, campaignId: made.campaignId, snapshotId: pf.snapshotId, children: before.children, noInspectionWrites: true }) });
  });
}
