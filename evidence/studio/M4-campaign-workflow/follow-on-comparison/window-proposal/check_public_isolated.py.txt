from pathlib import Path
import hashlib,json,os,subprocess,sys,time
root=Path.cwd();base=root/'.studio-local/comparison-next/window-proposal';checkout=root/'.studio-local/comparison-next/isolated-source';out=base/'public-isolated-successor';out.mkdir(exist_ok=False)
frozen=json.loads((root/'.studio-local/campaign-workflow/verified-targeted/source-before.json').read_text())
assert all(hashlib.sha256((root/n).read_bytes()).hexdigest()==h for n,h in frozen.items())
changed='services/studio-api/src/xverse_studio/comparison/runs.py'
files=[p.relative_to(root).as_posix() for p in sorted((root/'services/studio-api/src/xverse_studio').rglob('*.py'))]
assert all((root/n).read_bytes()==(checkout/n).read_bytes() for n in files)
(checkout/changed).write_bytes((base/'isolated/runs.py').read_bytes())
control={n:{'main':hashlib.sha256((root/n).read_bytes()).hexdigest(),'isolated':hashlib.sha256((checkout/n).read_bytes()).hexdigest()} for n in files}
assert [n for n,v in control.items() if v['main']!=v['isolated']]==[changed]
(out/'source-control.json').write_text(json.dumps({'scope':'All current backend package Python files match main except proposed window/observer validation in comparison/runs.py; native worker unchanged.','files':control},indent=2)+'\n')
invoke=out/'invoke.py';invoke.write_text('''import sys\nfrom pathlib import Path\nsys.path.insert(0,sys.argv[1])\nimport xverse_studio.api.advanced as advanced\nimport xverse_studio.comparison.runs as comparison\nassert Path(advanced.__file__).resolve().is_relative_to(Path(sys.argv[1]).resolve())\nassert Path(comparison.__file__).resolve().is_relative_to(Path(sys.argv[1]).resolve())\nimport pytest\nraise SystemExit(pytest.main(sys.argv[2:]))\n''')
command=[sys.executable,'-I',str(invoke),str(checkout/'services/studio-api/src'),str(base/'test_public_comparison_windows.py'),str(base/'test_comparison_windows.py'),'-q','-p','no:cacheprovider','--tb=short','--junitxml='+str(out/'junit.xml'),'--basetemp='+str(out/'tmp')]
env=dict(os.environ,PYTEST_DISABLE_PLUGIN_AUTOLOAD='1',TMPDIR=str(root.parent/'.xverse-studio-cache/tmp'));env.pop('PYTHONPATH',None)
started=time.monotonic()
with (out/'pytest.log').open('w') as log:r=subprocess.run(command,cwd=checkout,env=env,stdout=log,stderr=subprocess.STDOUT)
assert all(hashlib.sha256((root/n).read_bytes()).hexdigest()==h for n,h in frozen.items())
assert all(hashlib.sha256((checkout/n).read_bytes()).hexdigest()==v['isolated'] for n,v in control.items())
(out/'actual-window-projection.json').write_bytes((checkout/'.studio-local/comparison-next/window-proposal/actual-window-projection.json').read_bytes())
summary={'scope':'Isolated proposed comparator with actual public API/native reference runs plus8 labelled fixture cases; not applied to main, not881/fullT091 acceptance.','command':command,'exitCode':r.returncode,'seconds':round(time.monotonic()-started,3),'mainFrozenInputsUnchanged':True,'isolatedInputsUnchanged':True}
(out/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
print(json.dumps(summary),flush=True);print((out/'pytest.log').read_text()[-500:],flush=True)
raise SystemExit(r.returncode)
